# Projects Feature Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add a Projects section where 3D printing builds are tracked with files, links, markdown notes, and aggregated print history stats.

**Architecture:** Four new SQLite tables (`projects`, `project_files`, `project_links`, `project_prints`) backed by a FastAPI router. The frontend is Lit web components with a new `/projects` grid page and `/projects/:id` detail page. Print stats (duration, cost, filament) are cached in `project_prints` at link time to avoid live Moonraker/InfluxDB lookups on every load.

**Tech Stack:** FastAPI, aiosqlite, Lit 3, TypeScript, Vite, `marked` (npm, markdown rendering)

async def list_projects() -> list[dict[str, Any]]:
    async with _conn() as db:
        db.row_factory = aiosqlite.Row
        async with db.execute(
            """SELECT p.*,
                      COALESCE(COUNT(pp.id), 0)  AS print_count,
                      SUM(pp.duration_s)          AS total_duration_s,
                      SUM(pp.cost_total)           AS total_cost,
                      SUM(pp.filament_g)           AS total_filament_g
               FROM projects p
               LEFT JOIN project_prints pp ON pp.project_id = p.id
               GROUP BY p.id
               ORDER BY p.created_at DESC"""
        ) as cur:
            return [dict(r) for r in await cur.fetchall()]


async def get_project(project_id: int) -> dict[str, Any] | None:
    async with _conn() as db:
        db.row_factory = aiosqlite.Row
        async with db.execute(
            """SELECT p.*,
                      COALESCE(COUNT(pp.id), 0)  AS print_count,
                      SUM(pp.duration_s)          AS total_duration_s,
                      SUM(pp.cost_total)           AS total_cost,
                      SUM(pp.filament_g)           AS total_filament_g
               FROM projects p
               LEFT JOIN project_prints pp ON pp.project_id = p.id
               WHERE p.id = ?
               GROUP BY p.id""",
            (project_id,),
        ) as cur:
            row = await cur.fetchone()
            if not row:
                return None
            result = dict(row)

        async with db.execute(
            "SELECT * FROM project_files WHERE project_id = ? ORDER BY id",
            (project_id,),
        ) as cur:
            files = [dict(r) for r in await cur.fetchall()]

        for f in files:
            async with db.execute(
                """SELECT COALESCE(COUNT(id), 0) AS print_count,
                          SUM(duration_s)         AS total_duration_s,
                          SUM(cost_total)          AS total_cost,
                          SUM(filament_g)          AS total_filament_g
                   FROM project_prints WHERE project_file_id = ?""",
                (f["id"],),
            ) as cur:
                f.update(dict(await cur.fetchone()))

        async with db.execute(
            "SELECT * FROM project_links WHERE project_id = ? ORDER BY id",
            (project_id,),
        ) as cur:
            links = [dict(r) for r in await cur.fetchall()]

        async with db.execute(
            "SELECT * FROM project_prints WHERE project_id = ? ORDER BY id",
            (project_id,),
        ) as cur:
            prints = [dict(r) for r in await cur.fetchall()]

        result["files"] = files
        result["links"] = links
        result["prints"] = prints
        return result


async def create_project(name: str) -> dict[str, Any]:
    async with _conn() as db:
        db.row_factory = aiosqlite.Row
        cur = await db.execute(
            "INSERT INTO projects (name, notes, created_at) VALUES (?, NULL, ?)",
            (name, int(time.time())),
        )
        await db.commit()
        async with db.execute("SELECT * FROM projects WHERE id = ?", (cur.lastrowid,)) as c:
            row = dict(await c.fetchone())
    row.update({
        "print_count": 0, "total_duration_s": None,
        "total_cost": None, "total_filament_g": None,
        "files": [], "links": [], "prints": [],
    })
    return row


async def update_project(project_id: int, data: dict[str, Any]) -> dict[str, Any] | None:
    async with _conn() as db:
        sets = ", ".join(f"{k} = :{k}" for k in data)
        await db.execute(
            f"UPDATE projects SET {sets} WHERE id = :_id",
            {**data, "_id": project_id},
        )
        await db.commit()
    return await get_project(project_id)


async def delete_project(project_id: int) -> None:

        await db.commit()

        await db.commit()

```python
import math
import os
import uuid
from pathlib import Path
from typing import Any

from fastapi import APIRouter, File, HTTPException, UploadFile
from pydantic import BaseModel

from app import db
from app.config import settings
from app.influx import influx
from app.moonraker import moonraker

router = APIRouter(prefix="/api/projects", tags=["projects"])

_MM_TO_GRAMS = math.pi * (0.875 ** 2) * 1.24 * 0.001


class ProjectCreate(BaseModel):
    name: str


class ProjectUpdate(BaseModel):
    name: str | None = None
    notes: str | None = None


class FileAdd(BaseModel):
    type: str
    filename: str
    label: str | None = None


class LinkAdd(BaseModel):
    url: str
    label: str | None = None


class PrintLink(BaseModel):
    job_id: str
    project_file_id: int | None = None


@router.get("")
async def list_projects() -> list[dict[str, Any]]:
    return await db.list_projects()


@router.post("", status_code=201)
async def create_project(body: ProjectCreate) -> dict[str, Any]:
    return await db.create_project(body.name)


@router.get("/{project_id}")
async def get_project(project_id: int) -> dict[str, Any]:
    project = await db.get_project(project_id)
    if not project:
        raise HTTPException(404, "Project not found")
    return project


@router.patch("/{project_id}")
async def update_project(project_id: int, body: ProjectUpdate) -> dict[str, Any]:
    data = {k: v for k, v in body.model_dump().items() if v is not None}
    result = await db.update_project(project_id, data)
    if not result:
        raise HTTPException(404, "Project not found")
    return result


@router.delete("/{project_id}", status_code=204)
async def delete_project(project_id: int) -> None:
    project = await db.get_project(project_id)
    if not project:
        raise HTTPException(404, "Project not found")
    for f in project.get("files", []):
        if f["type"] == "stl":
            try:
                os.remove(f["filename"])
            except OSError:
                pass
    await db.delete_project(project_id)


@router.post("/{project_id}/files", status_code=201)
async def add_file(project_id: int, body: FileAdd) -> dict[str, Any]:
    if not await db.get_project(project_id):
        raise HTTPException(404, "Project not found")
    if body.type not in ("gcode", "stl"):
        raise HTTPException(422, "type must be 'gcode' or 'stl'")
    return await db.add_project_file(project_id, body.type, body.filename, body.label)


@router.post("/{project_id}/files/upload", status_code=201)
async def upload_stl(
    project_id: int,
    label: str | None = None,
    file: UploadFile = File(...),
) -> dict[str, Any]:
    if not await db.get_project(project_id):
        raise HTTPException(404, "Project not found")
    uploads_dir = Path(settings.uploads_dir)
    uploads_dir.mkdir(parents=True, exist_ok=True)
    stored_name = f"{uuid.uuid4().hex[:8]}_{file.filename}"
    stored_path = str(uploads_dir / stored_name)
    contents = await file.read()
    with open(stored_path, "wb") as fh:
        fh.write(contents)
    return await db.add_project_file(project_id, "stl", stored_path, label or file.filename)


@router.delete("/{project_id}/files/{file_id}", status_code=204)
async def delete_file(project_id: int, file_id: int) -> None:
    f = await db.get_project_file(file_id)
    if not f or f["project_id"] != project_id:
        raise HTTPException(404, "File not found")
    if f["type"] == "stl":
        try:
            os.remove(f["filename"])
        except OSError:
            pass
    await db.delete_project_file(file_id)


@router.post("/{project_id}/links", status_code=201)
async def add_link(project_id: int, body: LinkAdd) -> dict[str, Any]:
    if not await db.get_project(project_id):
        raise HTTPException(404, "Project not found")
    return await db.add_project_link(project_id, body.url, body.label)


@router.delete("/{project_id}/links/{link_id}", status_code=204)
async def delete_link(project_id: int, link_id: int) -> None:
    await db.delete_project_link(link_id)


@router.post("/{project_id}/prints", status_code=201)
async def link_print(project_id: int, body: PrintLink) -> dict[str, Any]:
    if not await db.get_project(project_id):
        raise HTTPException(404, "Project not found")
    duration_s: float | None = None
    filament_g: float | None = None
    cost_total: float | None = None
    try:
        job_data = await moonraker.get("/server/history/job", uid=body.job_id)
        job = job_data.get("job", {})
        duration_s = job.get("print_duration")
        filament_mm = job.get("filament_used") or 0.0
        filament_g = round(filament_mm * _MM_TO_GRAMS, 2)
        filename = job.get("filename", "")
        for h in influx.query_history(limit=500):
            if h.get("filename") == filename:
                cost_total = h.get("cost_total")
                break
    except Exception:
        pass
    return await db.add_project_print(
        project_id, body.project_file_id, body.job_id,
        duration_s, filament_g, cost_total,
    )


@router.delete("/{project_id}/prints/{job_id}", status_code=204)
async def unlink_print(project_id: int, job_id: str) -> None:
    await db.delete_project_print(project_id, job_id)


@router.get("/{project_id}/files/{file_id}/download")
async def download_stl(project_id: int, file_id: int) -> Any:
    from fastapi.responses import FileResponse
    f = await db.get_project_file(file_id)
    if not f or f["project_id"] != project_id or f["type"] != "stl":
        raise HTTPException(404, "File not found")
    if not os.path.exists(f["filename"]):
        raise HTTPException(404, "File missing from disk")
    return FileResponse(f["filename"], filename=Path(f["filename"]).name)
```

type Route = '/' | '/history' | '/spools' | '/projects' | `/projects/${string}`

class Router {
  private listeners: Array<(r: Route) => void> = []

  get current(): Route {
    const p = window.location.pathname
    if (p === '/history') return '/history'
    if (p === '/spools') return '/spools'
    if (p === '/projects') return '/projects'
    if (p.startsWith('/projects/')) return p as `/projects/${string}`
    return '/'
  }

  get projectId(): number | null {
    const p = window.location.pathname
    if (p.startsWith('/projects/')) {
      const id = parseInt(p.split('/')[2], 10)
      return isNaN(id) ? null : id
    }
    return null
  }

  navigate(path: Route): void {
    window.history.pushState({}, '', path)
    this.listeners.forEach(l => l(this.current))
  }

  listen(fn: (r: Route) => void): () => void {
    this.listeners.push(fn)
    window.addEventListener('popstate', () => fn(this.current))
    return () => {
      this.listeners = this.listeners.filter(l => l !== fn)
    }
  }
}

export const router = new Router()

declare global {
  interface Window { router: Router }
}
window.router = router

    this._draft = this._saved

    this.updateComplete.then(() => {
      const ta = this.querySelector<HTMLTextAreaElement>('textarea')

      ta.focus()
      ta.selectionStart = ta.selectionEnd = ta.value.length

    this._editing = false

        bubbles: true,
      }))

      this._saved = prev
      this._draft = prev
      this._error = true
      setTimeout(() => { this._error = false; this.requestUpdate() }, 2000)
    }
  }

import { customElement, state } from 'lit/decorators.js'
import type { Project } from '../types'
import { getProject, deleteProject, addProjectLink, deleteProjectLink } from '../api'
import { router } from '../router'
import '../components/markdown-editor'
import '../components/project-files-section'

function _fmtHours(s: number | null): string {
  if (!s) return '—'
  const h = Math.round(s / 3600)
  return h < 1 ? '<1h' : `${h}h`
}

function _linkHost(url: string): string {
  try { return new URL(url).hostname.replace('www.', '') } catch { return url }
}

@customElement('project-detail-page')
export class ProjectDetailPage extends LitElement {
  createRenderRoot() { return this }

  @state() project: Project | null = null
  @state() loading = true
  @state() error = false

  connectedCallback() {
    super.connectedCallback()
    const id = router.projectId
    if (id) this._load(id)
    else this.error = true
  }

  private async _load(id: number) {
    this.loading = true
    try {
      this.project = await getProject(id)
    } catch {
      this.error = true
    } finally {
      this.loading = false
    }
  }

  private async _delete() {
    if (!this.project) return
    if (!confirm(`Delete project "${this.project.name}"? This cannot be undone.`)) return
    await deleteProject(this.project.id)
    router.navigate('/projects')
  }

  private async _addLink() {
    if (!this.project) return
    const url = prompt('Enter URL:')
    if (!url?.trim()) return
    const label = prompt('Label (e.g. "printables", optional):') ?? undefined
    const link = await addProjectLink(this.project.id, { url: url.trim(), label: label || undefined })
    this.project = { ...this.project, links: [...(this.project.links ?? []), link] }
  }

  private async _removeLink(linkId: number) {
    if (!this.project) return
    await deleteProjectLink(this.project.id, linkId)
    this.project = { ...this.project, links: (this.project.links ?? []).filter(l => l.id !== linkId) }
  }

  private _stat(label: string, value: string) {
    return html`
      <div style="background: #1a1a1a; border: 1px solid #222; border-radius: 6px; padding: 12px; display: flex; flex-direction: column; gap: 3px;">
        <span style="font-size: 10px; color: #555;">${label}</span>
        <span style="font-size: 18px; font-weight: 600; color: #e0e0e0;">${value}</span>
      </div>
    `
  }

  render() {
    if (this.loading) return html`<div style="padding: 40px; text-align: center; color: #555;">Loading...</div>`
    if (this.error || !this.project) return html`<div style="padding: 40px; text-align: center; color: #555;">Project not found.</div>`

    const p = this.project
    return html`
      <div style="max-width: 760px; margin: 0 auto; padding: 24px; display: flex; flex-direction: column; gap: 14px;">

        <!-- header -->
        <div style="display: flex; justify-content: space-between; align-items: flex-start;">

          </div>

            style="background: transparent; border: 1px solid #33181888; color: #ef4444; border-radius: 6px; padding: 6px 12px; font-size: 11px; cursor: pointer;"
            @click=${this._delete}
          >delete</button>
        </div>

        <!-- stats bar -->
        <div style="display: grid; grid-template-columns: repeat(4, 1fr); gap: 8px;">
          ${this._stat('prints', String(p.print_count))}
          ${this._stat('total time', _fmtHours(p.total_duration_s))}
          ${this._stat('total cost', p.total_cost != null ? '€' + p.total_cost.toFixed(2) : '—')}
          ${this._stat('filament', p.total_filament_g != null ? p.total_filament_g.toFixed(0) + 'g' : '—')}
        </div>

        <!-- notes (markdown editor) -->
        <markdown-editor

          }}
        ></markdown-editor>

        <div style="background: #1a1a1a; border: 1px solid #222; border-radius: 8px; padding: 16px;">
          <div style="display: flex; justify-content: space-between; align-items: center; margin-bottom: 12px;">
            <span style="font-size: 10px; color: #555; text-transform: uppercase; letter-spacing: 0.05em;">Links</span>
